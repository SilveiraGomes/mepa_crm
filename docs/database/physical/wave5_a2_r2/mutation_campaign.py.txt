def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def once(t, old, new, n=1):
    assert t.count(old) == n, f'expected {n} occurrence(s) of {old[:80]!r}, found {t.count(old)}'
    return t.replace(old, new)


def contract(fn):
    def apply(t):
        c = json.loads(t)
        fn(c)
        return json.dumps(c, indent=2, ensure_ascii=False) + '\n'
    return apply


def set_flags(op, lock, commit):
    def f(c):
        for s in c['services']:
            for o in s['operations']:
                if o['operation'] == op:
                    if lock is None:
                        o.pop('child_safety_required', None)
                        o.pop('commit_time_child_safety_required', None)
                    else:
                        o['child_safety_required'] = lock
                        o['commit_time_child_safety_required'] = commit
    return f

def run_validator():
    r = subprocess.run(['node', str(VAL)], cwd=ROOT, capture_output=True, text=True, encoding='utf-8')
    text = (r.stdout or '') + (r.stderr or '')
    fails = [l.strip() for l in text.splitlines() if 'op.child_safety' in l or 'safety.' in l and 'FAIL' in l.upper()]
    status = next((l.strip() for l in text.splitlines() if 'status:' in l), '')
    return r.returncode, status, fails[:4], text

    before = {str(p.relative_to(ROOT)): hashlib.sha256(b).hexdigest() for p, b in originals.items()}
    row = {'id': mid, 'description': desc, 'expected': expected, 'files': list(before)}
    try:
        for p, fn in edits:
            mutated = fn(originals[p].decode('utf-8')).encode('utf-8')
            row.setdefault('mutation_applied', True)

            p.write_bytes(mutated)

        row.update(exit_code=code, status=status, evidence=fails)
    finally:
        for p, b in originals.items():

    after = {str(p.relative_to(ROOT)): sha(p) for p, _ in edits}
    row['restored_byte_identical'] = before == after
    row['result'] = 'DETECTED' if row.get('exit_code') != 0 else 'NOT_DETECTED'
    results.append(row)
    print(f"{mid:4} exit={row.get('exit_code')} {row['result']:12} restored={row['restored_byte_identical']} changed={row.get('changed')}  {desc[:78]}")
