    private function launch(array $jobs): WaveFourWorkerHarness
    {
        $h = new WaveFourWorkerHarness(self::SCRATCH, 'audit-worker.php', 70);
        try {
            foreach ($jobs as $i => $j) {
                $h->start($i, $j);
            }
            foreach ($jobs as $i => $_) {
                $h->waitForState($i, 'ready', 90);
            }
        } catch (\Throwable $e) {
            $h->close();
            throw $e;
        }
        return $h;
    }

    private function collect(WaveFourWorkerHarness $h, int $n): array
    {
        $out = [];
        try {
            for ($i = 0; $i < $n; $i++) {
                $w = $h->collect($i, 70);
                self::assertSame('', trim($w['stderr']), 'stderr: ' . $w['stderr']);
                self::assertSame(0, $w['exit'], $w['stdout']);
                self::assertTrue($w['done']);
                $r = json_decode(trim($w['stdout']), true, 512, JSON_THROW_ON_ERROR);
                self::assertSame(0, $r['tx'], 'no open transaction left behind');
                $out[] = $r;
            }
            return $out;
        } finally {
            $h->close();
        }
    }

    private function race(string $op, array $args, int $n, ?callable $perWorker = null): array
    {
        $jobs = [];
        for ($i = 0; $i < $n; $i++) {
            $jobs[] = ['op' => $op, 'args' => $perWorker ? $perWorker($i) + $args : $args];
        }
        $h = $this->launch($jobs);
        foreach ($jobs as $i => $_) {
            $h->signal($i, 'go');
        }
        return $this->collect($h, $n);
    }

    private function tally(array $results): array
    {
        $c = [];
        foreach ($results as $r) {
            $k = $r['result'] === 'OK' ? ($r['data']['outcome'] ?? 'OK') : $r['result'];
            $c[$k] = ($c[$k] ?? 0) + 1;
        }
        ksort($c);
        return $c;
    }

    // ---- E4: authority revoked while the operation is in flight (real DB lock wait, not sleeps) -----------------------
    private function waitForLockWait(WaveFourWorkerHarness $h, string $needle): void
    {
        $deadline = microtime(true) + 45;
        $schema = $this->db()->getDatabaseName();
        while (microtime(true) < $deadline) {
            foreach ($this->db()->select('SELECT p.info AS q FROM performance_schema.data_lock_waits w JOIN performance_schema.threads t ON t.thread_id = w.requesting_thread_id JOIN information_schema.processlist p ON p.id = t.processlist_id WHERE p.db = ?', [$schema]) as $row) {
                if (str_contains(strtolower((string) $row->q), $needle)) {
                    return;
                }
            }
            if (file_exists($h->stateFile(0, 'done'))) {
                self::fail('worker finished before reaching a real lock wait');
            }
            usleep(10000);
        }
        self::fail('no real lock wait observed on ' . $needle);
    }

    private function inFlight(int $unit, array $job, callable $mutate): array
    {
        $lock = self::connect()->getConnection();
        $lock->beginTransaction();
        $lock->table('organizational_units')->where('id', $unit)->lockForUpdate()->first();
        $h = $this->launch([$job]);
        try {
            $h->signal(0, 'go');
            $h->waitForState(0, 'started', 25);
            $this->waitForLockWait($h, 'audit_logs');       // worker passed provisional auth and its business INSERT
            $mutate();                                      // authority changes underneath (autocommit, must not block)
            $lock->rollBack();
        } catch (\Throwable $e) {
            if ($lock->transactionLevel()) {
                $lock->rollBack();
            }
            $h->close();
            throw $e;
        }
        return $this->collect($h, 1)[0];
    }

        return [

    /** @dataProvider revocations */
    public function test_e4_revocation_before_commit_aborts_grade_record(string $kind, string $expected): void
    {
        $g = $this->gradeWorld();
        $t = $g['teacher'];
        $link = $t['links'][0];
        $archiveBlocked = null;
        $mutate = match ($kind) {
            'none' => fn () => null,
            'grant' => fn () => $this->db()->statement("UPDATE user_role_scopes SET status = 'AUD_REVOKED' WHERE id = ?", [$link['link']]),
            'grant_end' => fn () => $this->db()->statement('UPDATE user_role_scopes SET ends_at = UTC_TIMESTAMP(6) WHERE id = ?', [$link['link']]),
            'role' => fn () => $this->db()->statement('UPDATE roles SET is_active = 0 WHERE id = ?', [$link['role']]),
            'perm' => fn () => $this->db()->statement('DELETE FROM role_permissions WHERE role_id = ?', [$link['role']]),
            'session' => fn () => $this->db()->statement('UPDATE auth_sessions SET revoked_at = UTC_TIMESTAMP(6) WHERE id = ?', [$t['session']]),
            // The in-flight grade INSERT holds a shared lock on users(id) through the graded_by FK, so archiving the actor
            // must WAIT for the operation (serialization), not slip in before the commit. Proven with a short lock timeout.
            'user' => function () use ($t, &$archiveBlocked) {
                $other = self::connect()->getConnection();
                $other->statement('SET SESSION innodb_lock_wait_timeout = 2');
                try {
                    $other->statement('UPDATE users SET archived_at = UTC_TIMESTAMP(6) WHERE id = ?', [$t['user']]);
                    $archiveBlocked = false;
                } catch (\Throwable $e) {
                    $archiveBlocked = str_contains($e->getMessage(), 'Lock wait timeout');
                }
            },
            'assignment' => fn () => $this->db()->statement("UPDATE class_instructors SET status = 'AUD_CI_OFF', ends_at = UTC_TIMESTAMP(6) WHERE class_id = ?", [$g['class']]),
            'instructor' => fn () => $this->db()->statement("UPDATE instructors SET status = 'AUD_I_OFF' WHERE person_id = ?", [$t['person']]),
        };
        $r = $this->inFlight($g['unit'], ['op' => 'record', 'args' => ['actor' => $t['user'], 'session' => $t['session'], 'attempt' => $g['attempt'], 'score' => '77']], $mutate);
        if ($kind === 'user') {
            self::assertTrue($archiveBlocked, 'archiving the actor waits for the in-flight operation (FK shared lock)');
            $expected = 'OK';
        }
        self::assertSame($expected, $r['result']);
        self::assertSame($expected === 'OK' ? 1 : 0, $this->rows('grades', ['attempt_id' => $g['attempt']]), 'business write committed only when still authorized');
        self::assertSame($expected === 'OK' ? 1 : 0, $this->rows('audit_logs', ['entity_type' => 'grades', 'unit_id' => $g['unit']]), 'audit exists only with the committed write');
    }

    public function test_child_authorization_lapse_and_consent_revocation_before_commit(): void
    {
        $w = $this->world();
        $t = $this->actor($w['unit'], ['ACADEMY_ATTENDANCE']);
        $this->teach($t, $w['class']);
        // (1) guardian authorization runs out (canonical DB clock) between the provisional stage and the commit
        $sess = $this->row('class_sessions', ['class_id' => $w['class'], 'status' => 'AUD_S_PLAN']);
        $c = $this->child($w['unit']);
        $st = $this->student($w['class'], 'AUD_E_LIVE', $c['child']);
        $this->db()->statement('UPDATE guardian_authorizations SET ends_at = DATE_ADD(UTC_TIMESTAMP(6), INTERVAL 12 SECOND) WHERE id = ?', [$c['auth']]);
        $expiry = (string) $this->db()->table('guardian_authorizations')->where('id', $c['auth'])->value('ends_at');
        $job = ['op' => 'attend', 'args' => ['actor' => $t['user'], 'session' => $t['session'], 'class_session' => $sess, 'enrollment' => $st['id'], 'status' => 'AUD_A_IN']];
        $r = $this->inFlight($w['unit'], $job, function () use ($expiry) {
            $deadline = microtime(true) + 60;
            while (DomainClock::now($this->db())->format('Y-m-d H:i:s.u') <= $expiry) {
                self::assertLessThan($deadline, microtime(true));
                usleep(20000);
            }
        });
        self::assertLessThan($expiry, $r['started'], 'the worker started while the cover was valid');
        self::assertSame(AcademyReason::GUARDIAN_AUTHORIZATION_INVALID, $r['result']);
        self::assertSame(0, $this->rows('academic_attendance', ['class_session_id' => $sess]));

        // (2) consent revocation cannot slip in between the late check and the commit: it must WAIT for the operation
        $c2 = $this->child($w['unit']);
        $st2 = $this->student($w['class'], 'AUD_E_LIVE', $c2['child']);
        $sess2 = $this->row('class_sessions', ['class_id' => $w['class'], 'status' => 'AUD_S_PLAN']);
        $job2 = ['op' => 'attend', 'args' => ['actor' => $t['user'], 'session' => $t['session'], 'class_session' => $sess2, 'enrollment' => $st2['id'], 'status' => 'AUD_A_IN']];
        $blocked = null;
        $r2 = $this->inFlight($w['unit'], $job2, function () use ($c2, &$blocked) {
            $other = self::connect()->getConnection();
            $other->statement('SET SESSION innodb_lock_wait_timeout = 2');
            try {
                $other->statement("UPDATE person_consents SET revoked_at = UTC_TIMESTAMP(6), status = 'SYNTHETIC_REVOKED' WHERE id = ?", [$c2['consent']]);
                $blocked = false;
            } catch (\Throwable $e) {
                $blocked = str_contains($e->getMessage(), 'Lock wait timeout');
            }
        });
        self::assertTrue($blocked, 'consent revocation must wait for the in-flight decision (row locked until commit)');
        self::assertSame('OK', $r2['result'], 'serialized: the operation validly committed BEFORE the revocation');
        // after the revocation the very next attempt is denied
        $this->db()->statement("UPDATE person_consents SET revoked_at = UTC_TIMESTAMP(6), status = 'SYNTHETIC_REVOKED' WHERE id = ?", [$c2['consent']]);
        $sess3 = $this->row('class_sessions', ['class_id' => $w['class'], 'status' => 'AUD_S_PLAN']);
        $this->deny(AcademyReason::CONSENT_REQUIRED, fn () => (new \App\Domain\Academy\AcademicAttendanceService($this->rt()))->record($t['user'], $t['session'], $sess3, $st2['id'], 'AUD_A_IN'));
    }
