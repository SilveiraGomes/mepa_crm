Each probe injects ONE violation into the audited code (my own text, not the executor's), runs the validator,
expects a non-zero exit with the matching check id, then restores the file. Repo is verified clean at the end."""
import subprocess, pathlib, shutil

if bak.exists():
    shutil.rmtree(bak)
shutil.copytree(dom, bak)


def restore():
    for f in bak.glob("*.php"):
        shutil.copy(f, dom / f.name)


def validator():
    p = subprocess.run(["node", "scripts/validate-wave5-application-contracts.cjs"], cwd=root, capture_output=True, text=True)
    failed = [l.strip()[2:].split(":")[0] for l in p.stderr.splitlines() if l.strip().startswith("- ")]
    return p.returncode, failed


def probe(name, file, old, new, expect):
    path = dom / file
    src = path.read_text(encoding="utf-8")
    assert old in src, f"anchor missing for {name}"
    path.write_text(src.replace(old, new, 1), encoding="utf-8")
    try:
        rc, failed = validator()
    finally:
        restore()
    ok = rc == 1 and any(f.startswith(expect) for f in failed)
    print(f"{'CAUGHT' if ok else 'MISSED':<7}{name:<60} rc={rc} checks={[f for f in failed if f.startswith(expect)][:2] or failed[:2]}")
    return ok
