    // ---- 34/35: audit atomicity, independent (own failing writer throwing a NON-domain exception) --------
    public function test_p_c10_instructor_assignment_audit_failure_leaves_no_orphan(): void
    {
        $w = $this->world();
        $m = $this->actor($w['unit'], ['ACADEMY_MANAGE']);
        $fresh = $this->row('people');
        $audits = (int) $this->db()->table('audit_logs')->count();
        foreach ([false, true] as $writeFirst) {
            $svc = new InstructorAssignmentService($this->rt(null, $this->failing($writeFirst)));
            try {
                $svc->assign($m['user'], $m['session'], $w['class'], $fresh);
                self::fail('assign must fail when the required audit fails');

                self::assertStringContainsString('AUDITOR-INJECTED', $e->getMessage());
            }
            self::assertSame(0, $this->rows('instructors', ['person_id' => $fresh]), 'orphan instructor profile (writeFirst=' . (int) $writeFirst . ')');

        }

    public function test_p_audit_failure_rolls_back_every_write_operation_class(): void
    {
        $g = $this->gradeWorld();
        $t = $g['teacher'];
        $bad = fn () => $this->rt(null, $this->failing(true));
        $expect = function (callable $fn) {
            try {
                $fn();
                self::fail('expected the injected audit failure');
            } catch (\RuntimeException) {
                self::assertTrue(true);
            }
        };
        // grade.record then grade.revise
        $expect(fn () => (new GradeService($bad()))->record($t['user'], $t['session'], $g['attempt'], '50'));
        self::assertSame(0, $this->rows('grades', ['attempt_id' => $g['attempt']]));
        (new GradeService($this->rt()))->record($t['user'], $t['session'], $g['attempt'], '50');
        $expect(fn () => (new GradeService($bad()))->revise($t['user'], $t['session'], $g['attempt'], 1, '60', 'motivo'));
        self::assertSame(1, $this->rows('grades', ['attempt_id' => $g['attempt']]), 'revision rolled back');
        // enrollment.create / transition
        $e = $this->actor($g['unit'], ['ACADEMY_ENROLL']);
        $p = $this->row('people');
        $expect(fn () => (new EnrollmentService($bad()))->enroll($e['user'], $e['session'], $g['class'], $p));
        self::assertSame(0, $this->rows('enrollments', ['person_id' => $p]));
        $pending = $this->student($g['class'], 'AUD_E_NEW');
        $expect(fn () => (new EnrollmentService($bad()))->transition($e['user'], $e['session'], $pending['id'], 'AUD_E_LIVE'));
        self::assertSame('AUD_E_NEW', $this->db()->table('enrollments')->where('id', $pending['id'])->value('status'));
        self::assertNull($this->db()->table('enrollments')->where('id', $pending['id'])->value('approved_by'));
        // attendance
        $att = $this->actor($g['unit'], ['ACADEMY_ATTENDANCE']);
        $this->teach($att, $g['class']);
        $sess = $this->row('class_sessions', ['class_id' => $g['class'], 'status' => 'AUD_S_PLAN']);
        $expect(fn () => (new AcademicAttendanceService($bad()))->record($att['user'], $att['session'], $sess, $g['st']['id'], 'AUD_A_IN'));
        self::assertSame(0, $this->rows('academic_attendance', ['class_session_id' => $sess]));
        // certificate.issue + revoke
        $this->policyJson($g['version'], ['criteria' => [['type' => 'ADMINISTRATIVE_APPROVAL']]]);
        $done = $this->student($g['class'], 'AUD_E_DONE');
        $c = $this->actor($g['unit'], ['ACADEMY_CERTIFY']);
        $file = $this->row('files', ['owner_unit_id' => $g['unit'], 'status' => 'AVAILABLE']);
        $expect(fn () => (new CertificateService($bad()))->issue($c['user'], $c['session'], $done['id'], $file));
        self::assertSame(0, $this->rows('certificates', ['enrollment_id' => $done['id']]));
        $ok = (new CertificateService($this->rt()))->issue($c['user'], $c['session'], $done['id'], $file);
        $expect(fn () => (new CertificateService($bad()))->revoke($c['user'], $c['session'], $ok['certificate_id'], 'motivo'));
        self::assertNull($this->db()->table('certificates')->where('id', $ok['certificate_id'])->value('revoked_at'), 'revocation rolled back');
    }

        $after = json_decode($au->after_metadata, true);

    // ---- 27: GRADES_VIEW least privilege ---------------------------------------------------------------
    public function test_p_grades_view_separation_and_no_indirect_grade_leak(): void
    {
        $g = $this->gradeWorld();
        $t = $g['teacher'];
        $svc = new GradeService($this->rt());
        $rec = $svc->record($t['user'], $t['session'], $g['attempt'], '61.5');
        // ASSESS-only and VIEW-only cannot read; GRADES_VIEW-only can read but cannot write
        $this->deny(AcademyReason::NOT_AUTHORIZED, fn () => $svc->history($t['user'], $t['session'], $g['attempt']));
        $viewOnly = $this->actor($g['unit'], ['ACADEMY_VIEW']);
        $this->deny(AcademyReason::NOT_AUTHORIZED, fn () => $svc->history($viewOnly['user'], $viewOnly['session'], $g['attempt']));
        $gv = $this->actor($g['unit'], ['ACADEMY_GRADES_VIEW']);
        self::assertCount(1, $svc->history($gv['user'], $gv['session'], $g['attempt']));
        $this->teach($gv, $g['class']);
        $this->deny(AcademyReason::NOT_AUTHORIZED, fn () => $svc->record($gv['user'], $gv['session'], $g['attempt'], '10'));
        // a legitimately pedagogical write flow still works without GRADES_VIEW (record/revise/finalize need no read)
        $rv = $svc->revise($t['user'], $t['session'], $g['attempt'], 1, '70', 'ajuste');
        self::assertSame(2, $rv['version']);
        // indirect leaks: no score in results/contexts of operations available to non-viewers
        $leaks = [];
        foreach ([$rec, $rv] as $out) {
            if (array_key_exists('score', $out)) {
                $leaks[] = 'result';
            }
        }
        $e = $this->deny(AcademyReason::STALE_WRITE, fn () => $svc->revise($t['user'], $t['session'], $g['attempt'], 1, '80', 'x'));
        if (array_key_exists('score', $e->context)) {
            $leaks[] = 'stale context';
        }
        $e2 = $this->deny(AcademyReason::STATE_POLICY_PENDING, fn () => $svc->finalize($t['user'], $t['session'], $g['attempt'], 2));
        if (array_key_exists('score', $e2->context)) {
            $leaks[] = 'pending context';
        }
        self::assertSame([], $leaks);
        // wrong-scope viewer
        $foreign = $this->actor($this->row('organizational_units'), ['ACADEMY_GRADES_VIEW']);
        $this->deny(AcademyReason::OUT_OF_SCOPE, fn () => $svc->history($foreign['user'], $foreign['session'], $g['attempt']));
    }

    // ---- 21-23: F-03 / D-09 / policy resolver ---------------------------------------------------------------
    public function test_p_f03_no_physical_default_and_values_never_drive_completion_or_certificate(): void
    {
        // (1) no physical DEFAULT on the four columns
        $defaults = $this->db()->select("SELECT COLUMN_NAME c, COLUMN_DEFAULT d, IS_NULLABLE n FROM information_schema.columns WHERE table_schema = DATABASE() AND table_name = 'assessments' AND column_name IN ('max_score','pass_score','max_attempts','weight')");
        self::assertCount(4, $defaults);
        foreach ($defaults as $d) {
            self::assertNull($d->d, $d->c . ' has a physical default');
            self::assertSame('NO', $d->n);
        }
        // (2) service refuses to create without all four (never fills them)
        $w = $this->world();
        $m = $this->actor($w['unit'], ['ACADEMY_MANAGE']);
        $svc = new AssessmentService($this->rt());
        foreach ([[null, '5', 1, '1'], ['10', null, 1, '1'], ['10', '5', null, '1'], ['10', '5', 1, null]] as [$mx, $ps, $ma, $wt]) {
            $this->deny(AcademyReason::POLICY_NOT_CONFIGURED, fn () => $svc->create($m['user'], $m['session'], $w['version'], null, 'P', $mx, $ps, $ma, $wt));
        }
        self::assertSame(0, $this->rows('assessments', ['course_version_id' => $w['version']]));
        // (3)/(4) the NOT NULL values never decide completion or certification: extreme pass_score/max_attempts/weight, same outcome
        $t = $this->actor($w['unit'], ['ACADEMY_ASSESS']);
        $this->teach($t, $w['class']);
        $st = $this->student($w['class']);
        $as = $this->assessment($w['version'], ['max_score' => '100', 'pass_score' => '99', 'max_attempts' => 1, 'weight' => '0.0001']);
        $att = $this->attempt($as, $st['id']);
        $this->row('grades', ['attempt_id' => $att, 'version' => 1, 'score' => '60', 'status' => 'AUD_G_FINAL', 'graded_by' => $t['user']]);
        $this->policyJson($w['version'], ['criteria' => [['type' => 'ASSESSMENT_MIN_SCORE', 'assessment_id' => $as, 'min_score' => '60', 'attempt_selection' => 'LATEST']]]);
        $done = (new CompletionService($this->rt()))->complete($t['user'], $t['session'], $st['id']);
        self::assertSame('AUD_E_DONE', $done['to'], 'policy min_score (60) decides, not assessments.pass_score (99)');
        // ...and the mirror image: pass_score 1 must not rescue a grade below the POLICY minimum
        $st2 = $this->student($w['class']);
        $att2 = $this->attempt($as, $st2['id']);
        $this->db()->table('assessments')->where('id', $as)->update(['pass_score' => '1']);
        $this->row('grades', ['attempt_id' => $att2, 'version' => 1, 'score' => '59.9999', 'status' => 'AUD_G_FINAL', 'graded_by' => $t['user']]);
        $this->deny(AcademyReason::COMPLETION_CRITERIA_NOT_MET, fn () => (new CompletionService($this->rt()))->complete($t['user'], $t['session'], $st2['id']));
        // certificate never consults pass_score: policy present + completed enrollment => issued regardless of any grade
        $this->policyJson($w['version'], ['criteria' => [['type' => 'ADMINISTRATIVE_APPROVAL']]]);
        $c = $this->actor($w['unit'], ['ACADEMY_CERTIFY']);
        $file = $this->row('files', ['owner_unit_id' => $w['unit'], 'status' => 'AVAILABLE']);
        self::assertSame(1, (new CertificateService($this->rt()))->issue($c['user'], $c['session'], $st['id'], $file)['version']);
        // attempt limit not enforced (max_attempts = 1, three attempts)
        $svcA = new AssessmentAttemptService($this->rt());
        $st3 = $this->student($w['class']);
        $ns = [];
        for ($i = 0; $i < 3; $i++) {
            $ns[] = $svcA->start($t['user'], $t['session'], $as, $st3['id'])['attempt_number'];
        }
        self::assertSame([1, 2, 3], $ns);
        // observation: max_score IS used as the upper bound of a stored score
        $st4 = $this->student($w['class']);
        $att4 = $this->attempt($this->assessment($w['version'], ['max_score' => '1', 'pass_score' => '1']), $st4['id']);
        $this->deny(AcademyReason::INVALID_INPUT, fn () => (new GradeService($this->rt()))->record($t['user'], $t['session'], $att4, '5'));
    }

    public function test_p_attendance_uses_academic_table_bulk_atomic_and_child_safety(): void
    {

        $this->teach($t, $w['class']);
        $sess = $this->row('class_sessions', ['class_id' => $w['class'], 'status' => 'AUD_S_PLAN']);
        $s1 = $this->student($w['class']);
        $s2 = $this->student($w['class']);
        $svc = new AcademicAttendanceService($this->rt());
        // atomic bulk: second entry invalid -> first not stored either
        $this->deny(AcademyReason::ATTENDANCE_STATUS_INVALID, fn () => $svc->recordBulk($t['user'], $t['session'], $sess, [$s1['id'] => 'AUD_A_IN', $s2['id'] => 'AUD_A_BOGUS']));

        // enrollment of ANOTHER class in the same bulk aborts everything
        $other = $this->world();
        $foreign = $this->student($other['class']);
        $this->deny(AcademyReason::CONTEXT_MISMATCH, fn () => $svc->recordBulk($t['user'], $t['session'], $sess, [$s1['id'] => 'AUD_A_IN', $foreign['id'] => 'AUD_A_IN']));

        // no class assignment -> denied
        $nobody = $this->actor($w['unit'], ['ACADEMY_ATTENDANCE']);
        $this->deny(AcademyReason::CLASS_ASSIGNMENT_REQUIRED, fn () => $svc->record($nobody['user'], $nobody['session'], $sess, $s1['id'], 'AUD_A_IN'));

        $this->deny(AcademyReason::CONSENT_REQUIRED, fn () => $svc->record($t['user'], $t['session'], $sess, $sm['id'], 'AUD_A_IN'));
        $rev = $this->child($w['unit']);
        $rev['svc']->revokeAuthorization($rev['staff']['user'], $rev['staff']['session'], $rev['auth'], 'x');

    // ---- 31/33: certificates + transcripts ---------------------------------------------------------------------------------
    public function test_p_certificate_prerequisites_and_revoke_semantics(): void
    {
        $w = $this->world();
        $assess = $this->actor($w['unit'], ['ACADEMY_ASSESS']);
        $this->teach($assess, $w['class']);
        $cert = $this->actor($w['unit'], ['ACADEMY_CERTIFY']);
        $done = $this->student($w['class'], 'AUD_E_DONE');
        $file = $this->row('files', ['owner_unit_id' => $w['unit'], 'status' => 'AVAILABLE']);
        $svc = new CertificateService($this->rt());
        $this->deny(AcademyReason::NOT_AUTHORIZED, fn () => $svc->issue($assess['user'], $assess['session'], $done['id'], $file));
        $this->deny(AcademyReason::POLICY_NOT_CONFIGURED, fn () => $svc->issue($cert['user'], $cert['session'], $done['id'], $file));
        $this->policyJson($w['version'], ['criteria' => [['type' => 'ADMINISTRATIVE_APPROVAL']]]);
        $live = $this->student($w['class']);
        $this->deny(AcademyReason::ENROLLMENT_NOT_COMPLETED, fn () => $svc->issue($cert['user'], $cert['session'], $live['id'], $file));
        $outsider = $this->actor($this->row('organizational_units'), ['ACADEMY_CERTIFY']);
        $this->deny(AcademyReason::OUT_OF_SCOPE, fn () => $svc->issue($outsider['user'], $outsider['session'], $done['id'], $file));
        $quarantined = $this->row('files', ['owner_unit_id' => $w['unit'], 'status' => 'QUARANTINED']);
        $this->deny(AcademyReason::FILE_NOT_AVAILABLE, fn () => $svc->issue($cert['user'], $cert['session'], $done['id'], $quarantined));
        $r = $svc->issue($cert['user'], $cert['session'], $done['id'], $file, 'ok');
        $this->deny(AcademyReason::CERTIFICATE_ALREADY_EXISTS, fn () => $svc->issue($cert['user'], $cert['session'], $done['id'], $file));
        $row = $this->db()->table('certificates')->where('id', $r['certificate_id'])->first();
        self::assertSame(hash('sha256', $r['token'], true), $row->token_hash);
        self::assertStringNotContainsString($r['token'], (string) $this->db()->table('audit_logs')->where('entity_type', 'certificates')->where('entity_id', $r['certificate_id'])->value('after_metadata'));
        // revoke: reason, actor, ASSESS cannot, no DELETE
        $this->deny(AcademyReason::NOT_AUTHORIZED, fn () => $svc->revoke($assess['user'], $assess['session'], $r['certificate_id'], 'motivo'));
        $this->deny(AcademyReason::REASON_REQUIRED, fn () => $svc->revoke($cert['user'], $cert['session'], $r['certificate_id'], ' '));
        $svc->revoke($cert['user'], $cert['session'], $r['certificate_id'], 'emitido por engano');
        $after = $this->db()->table('certificates')->where('id', $r['certificate_id'])->first();
        self::assertNotNull($after);
        self::assertNotNull($after->revoked_at);
        $ra = $this->db()->table('audit_logs')->where('entity_id', $r['certificate_id'])->where('action', 'certificate.revoked')->first();
        self::assertSame([$cert['user'], 'emitido por engano'], [(int) $ra->actor_id, $ra->reason]);
        self::assertSame(0, (int) $this->db()->selectOne("SELECT COUNT(*) c FROM information_schema.triggers WHERE trigger_schema = DATABASE()")->c);
    }

        }
    }
