        path.write_bytes(mutate(original.decode('utf-8')).encode('utf-8'))
        run = subprocess.run([PHP, 'vendor/phpunit/phpunit/phpunit', 'tests/DatabaseV2/AcademyRemediationTest.php', '--no-configuration', '--bootstrap', 'vendor/autoload.php'], cwd=API, env=env, capture_output=True, text=True)
